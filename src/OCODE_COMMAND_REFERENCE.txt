game -- the language-level statements, the built-in functions, and every

================================================================================
PART 1 - LANGUAGE STATEMENTS
================================================================================
These are the core building blocks of every OCode program. They are NOT
functions -- you call them as the first word on a line.

--------------------------------------------------------------------------------
1.1  VARIABLES & ASSIGNMENT
--------------------------------------------------------------------------------

let <name> = <expr>                  -- declares (or re-assigns) a variable
let <name> be <expr>                 -- natural-language variant of the above
   Example:
     let x = 5
     let y be 10
     let name = "Alice"
     let nums = [1, 2, 3]

<name> += <expr>                     -- compound assignment: add and store
<name> -= <expr>                     -- compound assignment: subtract and store
<name> *= <expr>                     -- compound assignment: multiply and store
<name> /= <expr>                     -- compound assignment: divide and store
<name> %= <expr>                     -- compound assignment: modulo and store
   Example:
     let i = 0
     i += 1           # i is now 1
     i *= 5           # i is now 5

--------------------------------------------------------------------------------
1.2  OUTPUT
--------------------------------------------------------------------------------

say <expr>                           -- prints expr followed by a newline
   Example:
     say "hello world"
     say 3 + 4         # prints 7
     say [1, 2, 3]     # prints [1, 2, 3]

say_inline <expr>                    -- prints expr WITHOUT a trailing newline
   Example:
     say_inline "loading... "
     say "done"        # output: loading... done

--------------------------------------------------------------------------------
1.3  INPUT
--------------------------------------------------------------------------------

ask <prompt>                         -- BLOCKING: prompts the user and returns
                                     -- the typed string. Execution pauses
                                     -- until the user presses Enter.
   Example:
     let name = ask "What is your name? "
     say "Hello, " + name + "!"

     let raw = ask "Enter a number: "
     let num = to_number of raw
     say num * 2

--------------------------------------------------------------------------------
1.4  CONDITIONALS
--------------------------------------------------------------------------------

if <cond> ... end                              -- simple if
if <cond> ... else ... end                     -- if/else
if <cond> ... else if <cond2> ... else ... end -- if / else-if / else
if <cond> ... elif <cond2> ... end             -- 'elif' is a synonym for 'else if'
   Example:
     let score = 85
     if score >= 90
         say "A"
     else if score >= 80
         say "B"
     else if score >= 70
         say "C"
     else
         say "below C"
     end

     let age = 16
     if age < 13
         say "child"
     elif age < 20
         say "teen"
     else
         say "adult"
     end

--------------------------------------------------------------------------------
1.5  LOOPS
--------------------------------------------------------------------------------

while <cond> ... end                            -- repeat while cond is true
   Example:
     let n = 0
     while n < 5
         say n
         let n = n + 1
     end

for each <item> in <list_or_dict> ... end      -- iterate items of a list,
                                               -- or keys of a dict
for each <key>, <value> in <dict> ... end      -- iterate key/value pairs of dict
   Example:
     let fruits = ["apple", "banana", "cherry"]
     for each fruit in fruits
         say fruit
     end

     let scores = dict alice is 10, bob is 5
     for each name, score in scores
         say name + ": " + to_string of score
     end

repeat <N> times ... end                       -- repeat a block N times
   Example:
     repeat 3 times
         say "spin"
     end

break                                           -- exit the enclosing loop
skip                                            -- skip to next loop iteration
                                                -- (like 'continue' in other langs)
   Example:
     let i = 0
     while i < 10
         let i = i + 1
         if i mod 2 == 0
             skip          # skip even numbers
         end
         if i == 7
             break         # stop at 7
         end
         say i
     end

--------------------------------------------------------------------------------
1.6  FUNCTIONS
--------------------------------------------------------------------------------

define <name> with <p1>, <p2>, ... ... end      -- define a function
define <name> ... end                          -- function with no parameters
return <expr>                                  -- return a value from a function
return                                         -- bare return (returns 0)

Call syntax:  <name> with <arg1>, <arg2>, ...  -- call a function with args
              <name>                            -- call a function with no args

   Example:
     define square with n
         return n * n
     end

     define factorial with n
         if n < 2
             return 1
         end
         return n * factorial with n - 1
     end

     define greet with name
         say "Hello, " + name + "!"
     end

     say square with 6       # 36
     say factorial with 5    # 120
     greet with "World"      # Hello, World!

--------------------------------------------------------------------------------
1.7  LIST / DICT MUTATION
--------------------------------------------------------------------------------

add <value> to <list_var>                      -- append value to list (in place)
   Example:
     let nums = [10, 20, 30]
     add 40 to nums
     say nums            # [10, 20, 30, 40]

remove at <index> from <list_var>              -- remove list element by index
   Example:
     remove at 0 from nums
     say nums            # [20, 30, 40]

<target> at <key> be <value>                   -- set dict entry or list element
                                               -- (mutates in place)
   Example:
     let scores = dict alice is 10
     scores at "alice" be 15
     scores at "dave" be 20
     say scores at "alice"   # 15

     let arr = [1, 2, 3]
     arr at 0 be 99
     say arr                 # [99, 2, 3]

--------------------------------------------------------------------------------
1.8  ERROR HANDLING
--------------------------------------------------------------------------------

try ... end                                    -- silently swallow any error
try ... catch <ERR> ... end                    -- catch the error message
try ... catch <ERR> ... else ... end           -- run 'else' if no error

error_message                                  -- message of the last caught error
error_code                                     -- numeric code of the last caught
                                               -- error (0 if none)
raise <message> [with <code>]                  -- throw a user-defined error

   Example:
     try
         let data = parse_json of "{not valid json"
         say "should not reach here"
     catch ERR
         say "caught: " + ERR
     end

     define safeDivide with a, b
         try
             return a / b
         catch ERR
             return 0
         end
     end

     say safeDivide with 10, 2     # 5
     say safeDivide with 10, 0     # 0

     if x < 0
         raise "x must be non-negative" with 42
     end

--------------------------------------------------------------------------------
1.9  PACKAGES (LIBRARIES)
--------------------------------------------------------------------------------

use <package_name>                             -- load a library written in OCode
                                               -- (registers its defines)

   Example:
     use ai
     use math
     use weather

     ai_connect with "YOUR_API_KEY", "gpt-4o-mini", "https://api.openai.com/v1", "openai"
     say ai_ask with "What is OCode in one sentence?"

See Part 4 for the full list of packages.

--------------------------------------------------------------------------------
1.10  PYTHON BRIDGE (pyoc)
--------------------------------------------------------------------------------

pyoc                                           -- start a raw Python block
...                                            -- raw Python lines go here
end                                            -- end the block

NOTE: pyoc is unsupported on mobile / OCDroid. It will fail with a friendly
message. Use it only in the desktop version of OCode.

--------------------------------------------------------------------------------
1.11  COMMENTS
--------------------------------------------------------------------------------

# ...                                          -- line comment
// ...                                         -- line comment (alternative)
/* ... */                                      -- block comment (can span lines)

   Example:
     # this is a comment
     let x = 5  // also a comment
     /* this
        is a
        multi-line comment */

--------------------------------------------------------------------------------
1.12  END-OF-BLOCK KEYWORDS
--------------------------------------------------------------------------------
All of the following are accepted as block terminators (synonyms for 'end'):

  end, endif, endelse, endwhile, endfor, endeach,
  endtry, endcatch, enddefine, endfunc, endfunction, endrepeat

You can also use two-word forms: 'end if', 'end while', 'end for', etc.

================================================================================
PART 2 - OPERATORS & SPECIAL SYNTAX
================================================================================

--------------------------------------------------------------------------------
2.1  ARITHMETIC
--------------------------------------------------------------------------------
+ - * / %                            -- standard math operators
                                       (+ also concatenates strings, merges
                                        lists, and merges dicts)
plus / minus / mod / remainder       -- English aliases for + - % %
divided by                           -- English alias for /

   Example:
     say 3 + 4           # 7
     say "a" + "b"       # "ab"
     say [1] + [2]       # [1, 2]
     say 17 mod 5        # 2
     say 17 divided by 5 # 3

--------------------------------------------------------------------------------
2.2  COMPARISON
--------------------------------------------------------------------------------
==  !=  <  >  <=  >=                 -- standard comparison operators
=  is                                -- aliases for ==
is not                               -- alias for !=
is less than                         -- alias for <
is greater than                      -- alias for >
is less than or equal to             -- alias for <=
is greater than or equal to          -- alias for >=

   Example:
     if x is greater than 5
         say "big"
     end
     if x is not equal to 99
         say "not 99"
     end

--------------------------------------------------------------------------------
2.3  LOGICAL
--------------------------------------------------------------------------------
and / or / not                       -- short-circuit logical operators
                                       (word forms; also && || ! in v18)

   Example:
     if x > 5 and x < 20
         say "in range"
     end
     if not done
         say "still working"
     end

--------------------------------------------------------------------------------
2.4  TERNARY (v18)
--------------------------------------------------------------------------------
<cond> ? <if_true> : <if_false>      -- inline conditional expression

   Example:
     let label = (x >= 0) ? "non-negative" : "negative"
     say label

--------------------------------------------------------------------------------
2.5  LITERALS & COLLECTIONS
--------------------------------------------------------------------------------
[expr, expr, ...]                    -- list literal (allows trailing comma)
   Example:
     let nums = [10, 20, 30]
     let mixed = [1, "two", true, [2, 3]]

{"k": v, "k2": v2}                   -- JSON-style dict literal
                                       (bare identifiers also allowed as keys)
dict <K> is <V>, <K> is <V>, ...     -- natural-language dict literal

   Example:
     let d1 = {"name": "Alice", "age": 30}
     let d2 = dict theme is "dark", font is "sans"

<container> at <key>                 -- index access (list[i], string[i],
                                       dict["k"]); also: '<container> at <i>'
   Example:
     say [10, 20, 30] at 1           # 20
     say "hello" at 0                # 'h'
     say (dict name is "Bob") at "name"   # Bob

length of <X>                        -- length of string/list/dict (special form)
   Example:
     say length of "hello"           # 5
     say length of [1, 2, 3]         # 3

--------------------------------------------------------------------------------
2.6  HEX & ESCAPES
--------------------------------------------------------------------------------
0x...                                -- hexadecimal integer literal
   Example:
     let mask = 0xFF
     let port = 0x1F90

String escapes: \n \r \t \0 \" \\

--------------------------------------------------------------------------------
2.7  FUNCTION CALL (POSTFIX 'with')
--------------------------------------------------------------------------------
<callable> with <a>, <b>, ...        -- call a function with arguments
<callable>                           -- call a function with no arguments

   Example:
     say square with 6
     say factorial with 5
     greet with "World"

--------------------------------------------------------------------------------
2.8  SPECIAL GLOBAL: args
--------------------------------------------------------------------------------
args                                 -- magic global list (command-line args).
                                       In OCDroid it's ["OCDroid"].
   Example:
     say length of args       # 1 inside OCDroid
     say args at 0            # "OCDroid"

PART 3 - BUILT-IN FUNCTIONS
================================================================================
These are available everywhere -- no `use` needed. Built-ins are the standard
library of the OCode language itself.

--------------------------------------------------------------------------------
3.1  CORE (string, list, math, type)
--------------------------------------------------------------------------------

ask <prompt>                        -- BLOCKING input; returns user's typed string
   Example:
     let name = ask "Your name? "

length of <X>                       -- length of a string / list / dict
   Example:
     say length of "hello"          # 5
     say length of [1,2,3]          # 3
     say length of (dict a is 1)    # 1

uppercase of <s>                    -- UPPERCASE a string
   Example:
     say uppercase of "hello"       # HELLO

lowercase of <s>                    -- lowercase a string
   Example:
     say lowercase of "HELLO"       # hello

trim of <s>                         -- strip leading/trailing whitespace
   Example:
     say trim of "  hi  "           # "hi"

split <s> by <sep>                  -- split string into list ("" -> char list)
   Example:
     say split "a,b,c" by ","       # ["a", "b", "c"]
     say split "abc" by ""          # ["a", "b", "c"]

join <list> by <sep>                -- join list into string with separator
   Example:
     say join ["a","b","c"] by "-"  # "a-b-c"

contains <container> <item>         -- bool: item in string/list/dict
contains <item> in <container>      -- alternate form (same result)
   Example:
     say contains "hello world" "world"   # true
     say contains [1,2,3] 2              # true
     say contains "alice" in (dict alice is 10)   # true

replace <str> from <old> to <new>   -- replace FIRST occurrence of old with new
   Example:
     say replace "I like cats" from "cats" to "dogs"

index of <item> in <container>      -- index of item in list/string (-1 if absent)
   Example:
     say index of "o" in "hello"    # 4

slice <lst> from <start> to <end>   -- sublist/substring [start, end)
   Example:
     say slice "hello world" from 6 to 11   # "world"
     say slice [1,2,3,4,5] from 1 to 4      # [2,3,4]

sort <list>                         -- sort ascending (numbers numerically;
                                       else lexicographic)
   Example:
     say sort [5, 1, 4, 2, 3]       # [1, 2, 3, 4, 5]

reverse <list_or_str>               -- reversed list, or reversed string
   Example:
     say reverse [1, 2, 3]          # [3, 2, 1]
     say reverse "abc"              # "cba"

abs of <x>                          -- absolute value
   Example:
     say abs of -42                 # 42

round of <x>                        -- round to nearest integer
   Example:
     say round of 3.7               # 4

floor of <x>                        -- floor (round down)
   Example:
     say floor of 3.9               # 3

ceil of <x>                         -- ceiling (round up)
   Example:
     say ceil of 3.1                # 4

sqrt of <x>                         -- square root (errors on negative)
   Example:
     say sqrt of 16                 # 4

power <base> to <exp>               -- base raised to exp
   Example:
     say power 2 to 10              # 1024

min of <a> [and <b>]                -- min of two numbers, or min of a list
   Example:
     say min of 5 and 3             # 3
     say min of [4, 1, 7, 2]        # 1

max of <a> [and <b>]                -- max of two numbers, or max of a list
   Example:
     say max of 5 and 3             # 5
     say max of [4, 1, 7, 2]        # 7

random                              -- random float in [0, 1)
   Example:
     say random                     # e.g. 0.7234

random from <lo> to <hi>            -- random float in [lo, hi)
   Example:
     let r = random from 1 to 100
     say r

to_number of <x>                    -- convert string/bool to number
   Example:
     say (to_number of "42") + 8    # 50

to_string of <x>                    -- convert any value to its string form
   Example:
     say to_string of 100 + " dollars"

to_bool of <x>                      -- convert to bool (truthiness)
   Example:
     say to_bool of 0               # false
     say to_bool of "yes"           # true

is_number of <x>                    -- true if x is a number
is_string of <x>                    -- true if x is a string
is_bool of <x>                      -- true if x is a bool
is_list of <x>                      -- true if x is a list
   Example:
     say is_number of 3.14          # true
     say is_string of "hi"          # true
     say is_list of [1,2,3]         # true
     say is_bool of true            # true

3.3  DICT / JSON / TIME (v12)
--------------------------------------------------------------------------------

keys of <dict>                       -- list of dict keys (insertion order)
   Example:
     say keys of (dict a is 1, b is 2)   # ["a", "b"]

has <key> in <dict>                  -- bool: does dict contain the key?
   Example:
     if has "alice" in scores
         say "alice is registered"
     end

range <from> to <to> [step <n>]     -- list of numbers in [from, to]
range from <X> to <Y>                -- alternate form
   Example:
     say range 1 to 5               # [1, 2, 3, 4, 5]
     say range 0 to 10 step 2       # [0, 2, 4, 6, 8, 10]
     say range 5 to 1 step -1       # [5, 4, 3, 2, 1]

now                                  -- current epoch seconds (number)
   Example:
     say now                        # e.g. 1723352400

today                                -- current local date as "YYYY-MM-DD"
   Example:
     say today                      # e.g. "2026-08-11"

clock                                -- current local time as "HH:MM:SS"
   Example:
     say clock                      # e.g. "14:23:01"

sleep <secs>                         -- sleep N seconds (no-op on interrupt)
   Example:
     sleep 2

parse_json of <s>                    -- parse JSON string into value
   Example:
     let data = parse_json of "{\"name\":\"Alice\",\"age\":30}"
     say data at "name"             # "Alice"

to_json of <v>                       -- serialize value to JSON string
   Example:
     let obj = dict name is "Bob", age is 25
     say to_json of obj             # {"name":"Bob","age":25}

dict <k> is <v>, <k> is <v>, ...     -- natural-language dict literal
   Example:
     let d = dict theme is "dark", font is "sans"
--------------------------------------------------------------------------------
3.4  HTTP & SOCKETS (v7-v9)
--------------------------------------------------------------------------------

http_get <url>                       -- HTTP GET, returns response body string
   Example:
     say http_get "https://httpbin.org/json"

http_post <url> with <body>          -- HTTP POST, returns response body string
   Example:
     say http_post "https://httpbin.org/post" with "name=OCode"

http_request <url> with <method> [, <body> [, <headers>]]
                                     -- full HTTP request;
                                     -- returns [status, body, headers] list

fetch <url>                          -- shorthand GET, returns body string
fetch <url> with <body>              -- shorthand POST, returns body string
fetch <url> with <method>, <body> [, <headers>]
                                     -- full request, returns [status, body, headers]
   Example:
     let body = fetch "https://httpbin.org/json"
     let r = fetch "https://httpbin.org/post" with "POST", "{}", []
     say r at 0                    # status code

socket <host> <port>                 -- open TCP socket; returns numeric handle
                                     -- variants: socket H with P, socket H, P, socket H port P
   Example:
     let s = socket "example.com" 80

send <handle> <msg>                  -- send data on socket
                                     -- variants: send H with M, send H, M
   Example:
     send s "GET / HTTP/1.0\r\nHost: example.com\r\n\r\n"

readline <handle>                    -- read one line (incl. \n) from socket
   Example:
     say readline s                  # e.g. "HTTP/1.0 200 OK\r\n"

receive <handle> [N]                 -- read up to N bytes (or all available)
   Example:
     say receive s 100               # read up to 100 bytes
     say receive s                   # read all available

close <handle>                       -- close a socket handle
   Example:
     close s

Tips for writing OCode:
  - Most statements read like English sentences. When in doubt, try the
    natural-language form first ("for each x in list", "say hello", "if
    x is greater than 5").
  - Use `use <package>` at the top of your file to import a library.
  - Inside try/catch, the error message is captured into your ERR variable.
  - The `args` global is ["OCDroid"] when run inside the app.
  - `length of X` works on strings, lists, and dicts.
  - `+` is overloaded: numbers add, strings concatenate, lists/dicts merge.
  - `at` is the universal index/key accessor: `nums at 0`, `dict at "k"`.
  - All block-ending variants are interchangeable: `end`, `endif`, `endwhile`,
    `endfor`, `endeach`, `endtry`, `enddefine`, `endfunc`, `endfunction`,
    `endrepeat`, `endelse`, `endcatch`. You can also use two-word forms
    like `end if`, `end while`.

Happy OCoding!
